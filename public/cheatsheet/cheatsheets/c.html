  68. Complete Dynamic Array Example

    #include <stdio.h>
#include <stdlib.h>

int main(void)
{
    int size = 5;

    int *numbers =
        malloc(
            size * sizeof(int)
        );

    if (numbers == NULL)
    {
        return 1;
    }

    for (int i = 0; i < size; i++)
    {
        numbers[i] =
            (i + 1) * 10;
    }

    for (int i = 0; i < size; i++)
    {
        printf(
            "%d\n",
            numbers[i]
        );
    }

    free(numbers);

    numbers = NULL;

    return 0;
}

  83. Simple Calculator

    #include <stdio.h>

int main(void)
{
    double a;
    double b;

    char operator;

    printf(
        "Enter expression: "
    );

    scanf(
        "%lf %c %lf",
        &a,
        &operator,
        &b
    );

    switch (operator)
    {
        case '+':
            printf(
                "%.2f\n",
                a + b
            );
            break;

        case '-':
            printf(
                "%.2f\n",
                a - b
            );
            break;

        case '*':
            printf(
                "%.2f\n",
                a * b
            );
            break;

        case '/':

            if (b != 0)
            {
                printf(
                    "%.2f\n",
                    a / b
                );
            }
            else
            {
                printf(
                    "Cannot divide by zero\n"
                );
            }

            break;

        default:
            printf(
                "Invalid operator\n"
            );
    }

    return 0;
}
