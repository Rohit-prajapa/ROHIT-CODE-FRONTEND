    CREATE TABLE students (

    id INT PRIMARY KEY,

    name VARCHAR(100),

    age INT

  8. Auto Increment

    CREATE TABLE students (

    id INT
        PRIMARY KEY
        AUTO_INCREMENT,

    age INT

  9. NOT NULL

    CREATE TABLE students (

    id INT PRIMARY KEY,

    name VARCHAR(100)
        NOT NULL,

    age INT

  10. UNIQUE Constraint

    CREATE TABLE users (

    id INT PRIMARY KEY,

    email VARCHAR(100)
        UNIQUE

);

  11. DEFAULT Constraint

    CREATE TABLE users (

    id INT PRIMARY KEY,

    name VARCHAR(100),

    country VARCHAR(50)
        DEFAULT 'India'

);

  12. CHECK Constraint

    CREATE TABLE students (

    id INT PRIMARY KEY,

    name VARCHAR(100),

    age INT CHECK (age >= 18)

);

  13. Foreign Key

    CREATE TABLE orders (

    order_id INT PRIMARY KEY,

    student_id INT,

    FOREIGN KEY (student_id)
        REFERENCES students(id)

);

  51. GROUP BY

    SELECT
    department,
    COUNT(*) AS total_students

FROM students

GROUP BY department;

  52. HAVING

    SELECT
    department,
    COUNT(*) AS total

FROM students

GROUP BY department

HAVING COUNT(*) > 5;

  53. WHERE vs HAVING

    SELECT
    department,
    AVG(marks)

FROM students

WHERE age >= 18

GROUP BY department

HAVING AVG(marks) > 70;

  54. INNER JOIN

    SELECT
    students.name,
    courses.course_name

FROM students

INNER JOIN courses

ON students.course_id
    = courses.id;

  55. LEFT JOIN

    SELECT
    students.name,
    courses.course_name

FROM students

LEFT JOIN courses

ON students.course_id
    = courses.id;

  56. RIGHT JOIN

    SELECT
    students.name,
    courses.course_name

FROM students

RIGHT JOIN courses

ON students.course_id
    = courses.id;

  57. FULL OUTER JOIN

    SELECT
    students.name,
    courses.course_name

FROM students

FULL OUTER JOIN courses

ON students.course_id
    = courses.id;

    Note: MySQL does not directly support FULL OUTER JOIN.

  58. SELF JOIN

    SELECT
    employee.name,
    manager.name AS manager

FROM employees employee

LEFT JOIN employees manager

ON employee.manager_id
    = manager.id;

  59. CROSS JOIN

    SELECT *

FROM students

CROSS JOIN courses;

  60. Subquery

    SELECT *

WHERE marks > (

    SELECT AVG(marks)
    FROM students

  61. Subquery with IN

FROM students

WHERE course_id IN (

    SELECT id
    FROM courses

    SELECT *
FROM students s

WHERE EXISTS (

    SELECT 1
    FROM courses c

        = s.course_id

);

  65. CASE Statement

    SELECT
    name,
    marks,

    CASE

        WHEN marks >= 90
            THEN 'A'

        WHEN marks >= 75
            THEN 'B'

        WHEN marks >= 60
            THEN 'C'

        ELSE 'D'

    END AS grade

FROM students;

WHERE id = 1;

SET balance = balance + 1000

WHERE id = 2;

  76. SAVEPOINT

    START TRANSACTION;


UPDATE students

SET marks = 90

WHERE id = 1;


SAVEPOINT update_one;


UPDATE students

SET marks = 80

WHERE id = 2;


ROLLBACK TO update_one;


COMMIT;

  79. Primary Key vs Foreign Key

    PRIMARY KEY

- Uniquely identifies row
- Cannot contain NULL
- Must be unique


FOREIGN KEY

- References another table
- Creates relationship
- Can contain duplicate values

  80. DELETE vs TRUNCATE vs DROP

    DELETE

Removes selected rows.


TRUNCATE

Removes all rows quickly.


DROP

Removes entire table
including structure.

  81. JOIN Quick Reference

    INNER JOIN

Returns matching rows
from both tables.


LEFT JOIN

Returns all left-table rows
plus matching right rows.


RIGHT JOIN

Returns all right-table rows
plus matching left rows.


FULL OUTER JOIN

Returns all rows
from both tables.


CROSS JOIN

Returns every possible
combination.


SELF JOIN

Joins a table
with itself.
