  57. Runtime Polymorphism

{

    {
        cout << "Animal";

}

  93. C++ Quick Revision

    #include <iostream>
#include <vector>
#include <algorithm>

using namespace std;


int add(int a, int b)
{
    return a + b;
}


int main()
{
    // Variable

    int age = 21;


    // Condition

    if (age >= 18)
    {
        cout << "Adult";
    }


    // Vector

    vector<int> numbers = {
        30,
        10,
        20
    };


    // Loop

    for (int number : numbers)
    {
        cout << number
             << " ";
    }


    // Sort

    sort(
        numbers.begin(),
        numbers.end()
    );


    // Function

    cout << add(10, 20);


    // Pointer

    int *ptr = &age;

    cout << *ptr;


    return 0;
}
