  66. useReducer

    import {
    useReducer
} from "react";


function reducer(
    state,
    action
) {

    switch (action.type) {

        case "increment":

            return {
                count:
                    state.count + 1
            };


        case "decrement":

            return {
                count:
                    state.count - 1
            };


        default:

            return state;
    }
}


function Counter() {

    const [
        state,
        dispatch
    ] = useReducer(
        reducer,
        { count: 0 }
    );


    return (
        <>

            <h2>
                {state.count}
            </h2>


            <button
                onClick={() =>
                    dispatch({
                        type:
                            "increment"
                    })
                }
            >
                +
            </button>

        </>
    );
}
