MongoDB Cheat Sheet

Complete MongoDB quick reference covering databases, collections, CRUD operations, operators, arrays, indexes, aggregation and Mongoose.

  1. Check MongoDB Version

    mongod --version

  2. Start MongoDB Shell

    mongosh

  3. Show Databases

    show dbs

  4. Create / Switch Database

    use college

  5. Current Database

    db

  6. Delete Database

    db.dropDatabase()

  7. Create Collection

    db.createCollection("students")

  8. Show Collections

    show collections

  9. Delete Collection

    db.students.drop()

  10. Insert One Document

    db.students.insertOne({
    name: "Rohit",
    age: 21,
    course: "B.Tech",
    city: "Pune"
})

  11. Insert Multiple Documents

    db.students.insertMany([
    {
        name: "Rohit",
        age: 21,
        marks: 90
    },
    {
        name: "Aman",
        age: 22,
        marks: 85
    },
    {
        name: "Priya",
        age: 20,
        marks: 95
    }
])

  12. Find All Documents

    db.students.find()

  13. Find One Document

    db.students.findOne()

  14. Find with Condition

    db.students.find({
    age: 21
})

  15. Find One with Condition

    db.students.findOne({
    name: "Rohit"
})

  16. Projection

    db.students.find(
    {},
    {
        name: 1,
        marks: 1,
        _id: 0
    }
)

  17. Greater Than

    db.students.find({
    marks: {
        $gt: 80
    }
})

  18. Greater Than or Equal

    db.students.find({
    marks: {
        $gte: 80
    }
})

  19. Less Than

    db.students.find({
    age: {
        $lt: 22
    }
})

  20. Less Than or Equal

    db.students.find({
    age: {
        $lte: 21
    }
})

  21. Not Equal

    db.students.find({
    city: {
        $ne: "Mumbai"
    }
})

  22. $in Operator

    db.students.find({
    age: {
        $in: [20, 21, 22]
    }
})

  23. $nin Operator

    db.students.find({
    age: {
        $nin: [18, 19]
    }
})

  24. $and Operator

    db.students.find({
    $and: [
        {
            age: {
                $gte: 18
            }
        },
        {
            marks: {
                $gte: 80
            }
        }
    ]
})

  25. $or Operator

    db.students.find({
    $or: [
        {
            city: "Pune"
        },
        {
            marks: {
                $gt: 90
            }
        }
    ]
})

  26. $not Operator

    db.students.find({
    age: {
        $not: {
            $gt: 21
        }
    }
})

  27. Check Field Exists

    db.students.find({
    email: {
        $exists: true
    }
})

  28. Regular Expression

    db.students.find({
    name: {
        $regex: "^R",
        $options: "i"
    }
})

  29. Update One Document

    db.students.updateOne(
    {
        name: "Rohit"
    },
    {
        $set: {
            marks: 95
        }
    }
)

  30. Update Multiple Documents

    db.students.updateMany(
    {
        city: "Pune"
    },
    {
        $set: {
            state: "Maharashtra"
        }
    }
)

  31. $unset Operator

    db.students.updateOne(
    {
        name: "Rohit"
    },
    {
        $unset: {
            city: ""
        }
    }
)

  32. $inc Operator

    db.students.updateOne(
    {
        name: "Rohit"
    },
    {
        $inc: {
            marks: 5
        }
    }
)

  33. $mul Operator

    db.products.updateOne(
    {
        name: "Laptop"
    },
    {
        $mul: {

        }
    }
)

  34. Rename Field

    db.students.updateMany(
    {},
    {
        $rename: {
            "course": "degree"
        }
    }

  35. Upsert

    db.students.updateOne(
    {
        name: "Rahul"
    },
    {
        $set: {
            age: 22
        }

    {
        upsert: true

  36. Delete One Document

    name: "Rohit"

  37. Delete Multiple Documents

    db.students.deleteMany({

        $lt: 40
    }

        $gte: 80

  46. Array Document

    db.students.insertOne({
    name: "Rohit",

    skills: [
        "HTML",
        "CSS",
        "JavaScript"

  48. $all Operator

    db.students.find({
    skills: {
        $all: [
            "HTML",
            "JavaScript"
        ]
    }
})

  49. $size Operator

    db.students.find({
    skills: {
        $size: 3
    }
})

  50. Add Value to Array

    db.students.updateOne(
    {
        name: "Rohit"
    },
    {
        $push: {
            skills: "React"
        }
    }
)

  51. $addToSet

    db.students.updateOne(
    {
        name: "Rohit"
    },
    {
        $addToSet: {
            skills: "React"
        }
    }
)

  52. Remove Value from Array

    db.students.updateOne(
    {
        name: "Rohit"
    },
    {
        $pull: {
            skills: "CSS"
        }
    }
)

    name: "Rohit",

    }

  60. Aggregation Pipeline

    db.students.aggregate([
    {
        $match: {
            marks: {
                $gte: 80
            }
        }
    }
])

  61. $group

    db.students.aggregate([
    {
        $group: {
            _id: "$city",

            totalStudents: {
                $sum: 1
            }
        }
    }
])

  62. Average with Aggregation

    db.students.aggregate([
    {
        $group: {
            _id: "$city",

            averageMarks: {
                $avg: "$marks"
            }
        }
    }
])

  63. Maximum Value

    db.students.aggregate([
    {
        $group: {
            _id: null,

            highestMarks: {
                $max: "$marks"
            }
        }
    }
])

  64. Minimum Value

    db.students.aggregate([
    {
        $group: {
            _id: null,

            lowestMarks: {
                $min: "$marks"
            }
        }
    }
])

  65. $project

    db.students.aggregate([
    {
        $project: {
            name: 1,
            marks: 1,
            _id: 0
        }
    }
])

  66. $sort in Aggregation

    db.students.aggregate([
    {
        $sort: {
            marks: -1
        }
    }
])

  67. $limit in Aggregation

    db.students.aggregate([
    {
        $sort: {
            marks: -1
        }
    },
    {
        $limit: 5
    }
])

  68. $unwind

    db.students.aggregate([
    {
        $unwind: "$skills"
    }
])

  69. $lookup — Join Collections

    db.students.aggregate([
    {
        $lookup: {
            from: "courses",

            localField:
                "courseId",

            foreignField:
                "_id",

            as:
                "courseDetails"
        }
    }
])

  70. Aggregation Example

    db.students.aggregate([
    {
        $match: {
            marks: {
                $gte: 60
            }
        }
    },

    {
        $group: {
            _id: "$city",

            averageMarks: {
                $avg: "$marks"
            },

            totalStudents: {
                $sum: 1
            }
        }
    },

    {
        $sort: {
            averageMarks: -1
        }
    }
])

  75. Install Mongoose

    npm install mongoose

  76. Connect MongoDB with Mongoose

    const mongoose =
    require("mongoose");


async function connectDB() {

    try {

        await mongoose.connect(
            process.env.MONGO_URI
        );

        console.log(
            "MongoDB Connected"
        );

    } catch (error) {

        console.error(
            error.message
        );

        process.exit(1);
    }
}


module.exports = connectDB;

  77. Mongoose Schema

    const mongoose =
    require("mongoose");


const studentSchema =
    new mongoose.Schema({

        name: {
            type: String,
            required: true
        },

        email: {
            type: String,
            required: true,
            unique: true
        },

        age: {
            type: Number
        },

        skills: [
            String
        ]

    }, {
        timestamps: true
    });

  78. Mongoose Model

    const Student =
    mongoose.model(
        "Student",
        studentSchema
    );


module.exports = Student;
